# Connects your GitHub account with a temporary gist. Your GitHub login is never sent to us.
TF_ID="$HOME/.config/tokenfund/key"
if [ ! -s "$TF_ID" ]; then
  NONCE="$(curl -fsS -X POST "$API/api/id/start")" || die "The site can't be reached."
  GIST_URL="$(printf '%s\n' "$NONCE" | gh gist create -f tokenfund-proof.txt - 2>/dev/null)" || die "Please run: gh auth refresh -s gist. Then try again."
  GIST="${GIST_URL##*/}"
  sleep 2
  KEY="$(curl -fsS -X POST "$API/api/id/verify?gist=$GIST&nonce=$NONCE&login=$(gh api user -q .login)")"; RC=$?
  gh api -X DELETE "gists/$GIST" >/dev/null 2>&1
  [ "$RC" = 0 ] && [ -n "$KEY" ] || die "Could not verify your GitHub account. Try again in a minute."
  mkdir -p "$(dirname "$TF_ID")"; (umask 077; printf '%s' "$KEY" > "$TF_ID")
  say "GitHub account connected."
fi
TF_KEY="$(cat "$TF_ID")"
